## Step 1: Hello World

### 学习目标
创建你的第一个 Eino 程序，理解基本结构。

### 前置条件
- Go 1.21+
- 一个 LLM API Key（OpenAI 或 Ark）

### 完整代码

创建文件 `hello_eino.go`：

```go
package main

import (
    "context"
    "fmt"
    "log"

    "github.com/cloudwego/eino/components/model"
    "github.com/cloudwego/eino/components/model/openai"
    "github.com/cloudwego/eino/schema"
)

func main() {
    // 1. 创建 context（用于控制超时、取消等）
    ctx := context.Background()

    // 2. 创建 ChatModel（这里用 OpenAI，你可以换成其他模型）
    model, err := openai.NewChatModel(ctx, &openai.ChatModelConfig{
        Model: "gpt-4o-mini",                          // 模型名称
        APIKey: "your-api-key",                         // API Key
    })
    if err != nil {
        log.Fatal("创建模型失败: ", err)
    }

    // 3. 构造消息列表（可以有多个消息）
    messages := []*schema.Message{
        schema.SystemMessage("你是一个友好的助手"),      // 系统消息（可选）
        schema.UserMessage("你好，请用一句话介绍自己"), // 用户消息
    }

    // 4. 调用模型生成回复
    output, err := model.Generate(ctx, messages)
    if err != nil {
        log.Fatal("调用模型失败: ", err)
    }

    // 5. 打印结果
    fmt.Println("AI 回复: ", output.Content)
}
```

## Step 2: ChatModel

### 学习目标
深入理解 ChatModel 的配置和多种模型的使用。

### 完整代码

```go
package main

import (
    "context"
    "fmt"
    "log"
    "os"

    "github.com/cloudwego/eino/components/model"
    "github.com/cloudwego/eino/components/model/ark"
    "github.com/cloudwego/eino/components/model/openai"
    "github.com/cloudwego/eino/compose"
    "github.com/cloudwego/eino/schema"
)

// getModel 根据类型返回不同的 ChatModel
func getModel(ctx context.Context, modelType string) (model.ChatModel, error) {
    switch modelType {
    case "openai":
        return openai.NewChatModel(ctx, &openai.ChatModelConfig{
            Model:  os.Getenv("OPENAI_MODEL"),
            APIKey: os.Getenv("OPENAI_API_KEY"),
        })
    case "ark":
        return ark.NewChatModel(ctx, &ark.ChatModelConfig{
            Model:  os.Getenv("ARK_MODEL"),
            APIKey: os.Getenv("ARK_API_KEY"),
        })
    default:
        return nil, fmt.Errorf("未知的模型类型: %s", modelType)
    }
}

func main() {
    ctx := context.Background()

    // 从环境变量获取模型类型（默认 openai）
    modelType := os.Getenv("MODEL_TYPE")
    if modelType == "" {
        modelType = "openai"
    }

    // 创建模型
    model, err := getModel(ctx, modelType)
    if err != nil {
        log.Fatal(err)
    }

    // 带配置的调用
    messages := []*schema.Message{
        schema.UserMessage("解释什么是量子计算，用一句话"),
    }

    // 使用 Option 配置调用参数
    output, err := model.Generate(ctx, messages,
        compose.WithTemperature(0.7),  // 温度：控制随机性（0-2，越高越随机）
        compose.WithMaxTokens(100),     // 最大 Token 数
    )
    if err != nil {
        log.Fatal(err)
    }

    fmt.Printf("回复: %s\n", output.Content)
    fmt.Printf("Token 使用: %d in / %d out\n",
        output.ResponseMeta.Usage.InputTokens,
        output.ResponseMeta.Usage.OutputTokens)
}
```

| 配置项 | 说明 | 示例值 |
|--------|------|--------|
| `Model` | 模型名称 | `"gpt-4o"`, `"ep-xxx"` |
| `APIKey` | API 密钥 | `"sk-xxx"` |
| `BaseURL` | API 地址 | `"https://api.openai.com/v1"` |

### 调用选项

```go
// 常用选项
compose.WithTemperature(0.7)     // 温度：0-2，默认0.7
compose.WithTopP(0.9)           // Top-P 采样：0-1
compose.WithMaxTokens(1000)     // 最大 Token 数
compose.WithModel("gpt-4")      // 覆盖默认模型
compose.WithTimeout(30*time.Second)  // 超时时间
```

## Step 3: Message

### 学习目标
掌握 Message 的各种构造方式和多模态支持。

### Message 结构详解

```go
type Message struct {
    Role       RoleType    // 角色：system/user/assistant/tool
    Content    string      // 文本内容

    // 多模态输入（支持图片等）
    UserInputMultiContent []MessageInputPart

    // 多模态输出
    AssistantGenMultiContent []MessageOutputPart

    // 工具调用（assistant 消息使用）
    ToolCalls  []ToolCall
    ToolCallID string
    ToolName   string

    // 元数据
    ResponseMeta    *ResponseMeta   // Token 使用量等
    ReasoningContent string         // 推理内容（如 o1 模型）
}
```

### 完整代码

```go
package main

import (
    "context"
    "fmt"
    "log"

    "github.com/cloudwego/eino/components/model/openai"
    "github.com/cloudwego/eino/schema"
)

func main() {
    ctx := context.Background()
    model, _ := openai.NewChatModel(ctx, &openai.ChatModelConfig{
        Model:  "gpt-4o-mini",
        APIKey: "your-key",
    })

    // ============ 1. 简单消息 ============
    simple := schema.UserMessage("Hello")
    fmt.Printf("简单用户消息: %+v\n", simple)
    // 输出: Role:user, Content:Hello

    // ============ 2. 带图片的消息 ============
    withImage := &schema.Message{
        Role: schema.User,
        UserInputMultiContent: []schema.MessageInputPart{
            &schema.TextContent{Text: "这张图片里有什么？"},
            &schema.ImageURLContent{URL: "https://example.com/image.jpg"},
        },
    }

    // ============ 3. 工具调用消息 ============
    toolMessage := schema.ToolMessage(
        `{"temp": 25, "weather": "晴"}`,  // 工具返回的 JSON
        "call_123",                         // 工具调用 ID
    )
    fmt.Printf("工具消息: %+v\n", toolMessage)
    // 输出: Role:tool, Content:{...}, ToolCallID:call_123

    // ============ 4. 多轮对话 ============
    conversation := []*schema.Message{
        schema.SystemMessage("你是一个诗人"),
        schema.UserMessage("写一首关于春天的诗"),
        schema.AssistantMessage("春风又绿江南岸，\n明月何时照我还。", nil),
        schema.UserMessage("这首诗是谁写的？"),
    }

    output, err := model.Generate(ctx, conversation)
    if err != nil {
        log.Fatal(err)
    }
    fmt.Printf("AI 回复: %s\n", output.Content)

    // ============ 5. 提取对话历史 ============
    // 通常从 Memory 中获取，这里演示结构
    history := conversation
    history = append(history, output)  // 添加 AI 回复
    fmt.Printf("对话历史共 %d 条消息\n", len(history))
}
```

### Role 类型

| Role | 说明 | 示例 |
|------|------|------|
| `System` | 系统指令 | 设定 AI 角色 |
| `User` | 用户输入 | 用户的问题 |
| `Assistant` | AI 回复 | AI 生成的内容 |
| `Tool` | 工具结果 | 工具执行后返回 |

## Step 4: Stream

### 学习目标
理解流式输出的原理和实现。

### 为什么需要流式

| 模式 | 特点 | 用户体验 |
|------|------|----------|
| 非流式 | 等待全部生成 | 等待时间长 |
| 流式 | 逐步返回 | 实时显示，像打字 |

### 完整代码

```go
package main

import (
    "context"
    "errors"
    "fmt"
    "io"
    "log"

    "github.com/cloudwego/eino/components/model/openai"
    "github.com/cloudwego/eino/schema"
)

func main() {
    ctx := context.Background()
    model, _ := openai.NewChatModel(ctx, &openai.ChatModelConfig{
        Model:  "gpt-4o-mini",
        APIKey: "your-key",
    })

    messages := []*schema.Message{
        schema.UserMessage("给我讲一个关于机器人的故事，50字左右"),
    }

    // ============ 流式调用 ============
    stream, err := model.Stream(ctx, messages)
    if err != nil {
        log.Fatal("Stream 失败: ", err)
    }
    defer stream.Close()  // 记得关闭流

    fmt.Println("AI: ", end="")  // 不换行

    // 逐步读取
    for {
        chunk, err := stream.Recv()
        if errors.Is(err, io.EOF) {
            // 流结束
            fmt.Println()  // 换行
            break
        }
        if err != nil {
            log.Fatal("读取失败: ", err)
        }

        // 打印内容（不换行）
        fmt.Print(chunk.Content)
    }

    // ============ 流式信息 ============
    // 如果需要获取完整响应
    stream2, _ := model.Stream(ctx, messages)

    var fullContent string
    for {
        chunk, err := stream2.Recv()
        if errors.Is(err, io.EOF) {
            break
        }
        fullContent += chunk.Content
    }

    fmt.Printf("\n完整内容长度: %d 字符\n", len(fullContent))
}
```

### StreamReader 核心方法

```go
// Recv() 接收一个数据块
chunk, err := stream.Recv()

// Close() 关闭流
stream.Close()

// 常用模式
for {
    chunk, err := stream.Recv()
    if errors.Is(err, io.EOF) {
        break
    }
    if err != nil {
        // 处理错误
    }
    // 使用 chunk
}
```

### 流的管道操作

```go
// 创建管道（生产者-消费者）
sr, sw := schema.Pipe[string](10)  // buffer size

// 生产者
go func() {
    for _, s := range data {
        sw.Send(s)  // 发送数据
    }
    sw.Close()  // 关闭流
}()

// 消费者
for {
    item, err := sr.Recv()
    if errors.Is(err, io.EOF) {
        break
    }
    fmt.Println(item)
}

// 合并多个流
merged := schema.MergeStreamReaders(stream1, stream2)

// 带转换的流
converted := schema.StreamReaderWithConvert(stream, func(in string) (int, error) {
    return len(in), nil
})
```

## Step 5: Tool

### 学习目标
创建自定义工具，让 AI 能够调用函数。

### 什么是 Tool

Tool 是让 AI 能够执行实际操作的能力，比如：
- 搜索信息
- 查询天气
- 执行计算
- 读写文件

### 完整代码

```go
package main

import (
    "context"
    "fmt"
    "log"

    "github.com/cloudwego/eino/components/model/openai"
    "github.com/cloudwego/eino/components/tool/utils"
    "github.com/cloudwego/eino/compose"
    "github.com/cloudwego/eino/schema"
)

// ============ 定义工具请求结构 ============
type CalculatorRequest struct {
    A int `json:"a"`  // 第一个数
    B int `json:"b"`  // 第二个数
    Op string `json:"op"` // 操作符：add, sub, mul, div
}

// ============ 创建计算器工具 ============
func NewCalculatorTool() (tool.BaseTool, error) {
    return utils.InferTool(
        "calculator",                      // 工具名称
        "执行数学计算，支持加减乘除",       // 工具描述（AI 会看到这个）
        calculatorImpl,                     // 实现函数
    )
}

// ============ 工具实现函数 ============
var calculatorImpl = func(ctx context.Context, req *CalculatorRequest) (string, error) {
    var result int

    switch req.Op {
    case "add":
        result = req.A + req.B
    case "sub":
        result = req.A - req.B
    case "mul":
        result = req.A * req.B
    case "div":
        if req.B == 0 {
            return "错误：除数不能为0", nil
        }
        result = req.A / req.B
    default:
        return fmt.Sprintf("未知操作: %s", req.Op), nil
    }

    return fmt.Sprintf("%d", result), nil
}

// ============ 创建搜索工具 ============
func NewSearchTool() (tool.BaseTool, error) {
    return utils.InferTool(
        "web_search",
        "搜索互联网获取最新信息",
        func(ctx context.Context, req *SearchRequest) (string, error) {
            // 实际项目中调用搜索 API
            return fmt.Sprintf("搜索结果：关于「%s」的最热内容是...", req.Query), nil
        },
    )
}

type SearchRequest struct {
    Query string `json:"query"`
}

func main() {
    ctx := context.Background()

    // 创建模型
    model, _ := openai.NewChatModel(ctx, &openai.ChatModelConfig{
        Model:  "gpt-4o",
        APIKey: "your-key",
    })

    // ============ 创建工具 ============
    calcTool, err := NewCalculatorTool()
    if err != nil {
        log.Fatal(err)
    }

    searchTool, err := NewSearchTool()
    if err != nil {
        log.Fatal(err)
    }

    // ============ 使用工具的 ChatModel ============
    chatModel, err := model.WithTools([]*schema.ToolInfo{
        // 从工具获取信息
        {Name: "calculator", Desc: "执行数学计算，支持加减乘除",
         ArgsSchema: &CalculatorRequest{}},
        {Name: "web_search", Desc: "搜索互联网获取最新信息",
         ArgsSchema: &SearchRequest{}},
    })
    if err != nil {
        log.Fatal(err)
    }

    // ============ 测试工具调用 ============
    messages := []*schema.Message{
        schema.UserMessage("请帮我计算 123 + 456 等于多少？"),
    }

    output, err := chatModel.Generate(ctx, messages)
    if err != nil {
        log.Fatal(err)
    }

    // 检查是否有工具调用
    if len(output.ToolCalls) > 0 {
        fmt.Printf("AI 调用了工具: %s\n", output.ToolCalls[0].Function.Name)
        fmt.Printf("参数: %s\n", output.ToolCalls[0].Function.Arguments)
    } else {
        fmt.Printf("AI 回复: %s\n", output.Content)
    }
}
```

### 工具结构

```
┌─────────────────────────────────────────────────────────────────┐
│                      Tool 调用流程                                │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  1. 定义工具                                                    │
```

## Step 6: Agent

### 学习目标
创建 ReAct Agent，实现"思考+行动"的智能体。

### 什么是 ReAct

ReAct = Reasoning + Acting（推理 + 行动）

让 AI 能够：思考 → 决定使用工具 → 执行 → 观察结果 → 继续思考

### 完整代码

```go
package main

import (
    "context"
    "errors"
    "fmt"
    "io"
    "log"

    "github.com/cloudwego/eino/components/model/openai"
    "github.com/cloudwego/eino/components/tool/utils"
    "github.com/cloudwego/eino/compose"
    "github.com/cloudwego/eino/flow/agent/react"
    "github.com/cloudwego/eino/schema"
)

type SearchRequest struct {
    Query string `json:"query"`
}

func main() {
    ctx := context.Background()

    // ============ 1. 创建模型 ============
    model, _ := openai.NewChatModel(ctx, &openai.ChatModelConfig{
        Model:  "gpt-4o",
        APIKey: "your-key",
    })

    // ============ 2. 创建工具 ============
    searchTool, _ := utils.InferTool(
        "search",
        "搜索互联网获取最新信息",
        func(ctx context.Context, req *SearchRequest) (string, error) {
            // 实际项目中调用搜索 API
            return fmt.Sprintf("[搜索结果] 关于「%s」的信息：北京今天晴，28度", req.Query), nil
        },
    )

    // ============ 3. 创建 ReAct Agent ============
    agent, err := react.NewAgent(ctx, &react.AgentConfig{
        Model: model,
        ToolsConfig: compose.ToolsNodeConfig{
            Tools: []tool.BaseTool{searchTool},
        },
        MaxStep: 10,  // 最大迭代次数，防止死循环
    })
    if err != nil {
        log.Fatal("创建 Agent 失败: ", err)
    }

    // ============ 4. 执行 Agent ============
    messages := []*schema.Message{
        schema.SystemMessage("你是一个有帮助的助手，可以使用搜索工具来获取最新信息。"),
        schema.UserMessage("北京今天的天气怎么样？"),
    }

    stream, err := agent.Stream(ctx, messages)
    if err != nil {
        log.Fatal("Agent 执行失败: ", err)
    }
    defer stream.Close()

    // ============ 5. 处理流式输出 ============
    fmt.Println("AI: ")
    for {
        chunk, err := stream.Recv()
        if errors.Is(err, io.EOF) {
            break
        }
        if err != nil {
            log.Fatal("读取失败: ", err)
        }

        // 打印思考过程和回复
        if chunk.Content != "" {
            fmt.Print(chunk.Content)
        }

        // 显示工具调用
        for _, tc := range chunk.ToolCalls {
            fmt.Printf("\n[思考] 我应该调用 %s 工具\n", tc.Function.Name)
        }
    }
    fmt.Println()
}
```

## Step 7: Memory

### 学习目标
实现对话历史的存储和管理。

### 为什么需要 Memory

| 没有 Memory | 有 Memory |
|-------------|-----------|
| 每次都是新对话 | 记得之前的对话 |
| 无法上下文 | 理解对话连续性 |

### 完整代码

```go
package main

import (
    "context"
    "fmt"
    "log"
    "sync"

    "github.com/cloudwego/eino/schema"
)

// ============ 简单 Memory 实现 ============
type SimpleMemory struct {
    mu           sync.RWMutex
    conversations map[string][]*schema.Message  // 对话 ID -> 消息列表
    maxWindow    int                           // 最大保留消息数
}

func NewSimpleMemory(maxWindow int) *SimpleMemory {
    return &SimpleMemory{
        conversations: make(map[string][]*schema.Message),
        maxWindow:    maxWindow,
    }
}

// 获取对话历史
func (m *SimpleMemory) GetMessages(conversationID string) []*schema.Message {
    m.mu.RLock()
    defer m.mu.RUnlock()

    msgs := m.conversations[conversationID]
    // 只返回最近的消息（滑动窗口）
    if len(msgs) > m.maxWindow {
        return msgs[len(msgs)-m.maxWindow:]
    }
    return msgs
}

// 添加消息
func (m *SimpleMemory) Append(conversationID string, msg *schema.Message) {
    m.mu.Lock()
    defer m.mu.Unlock()

    m.conversations[conversationID] = append(
        m.conversations[conversationID],
        msg,
    )
}

// 清除对话
func (m *SimpleMemory) Clear(conversationID string) {
    m.mu.Lock()
    defer m.mu.Unlock()
    delete(m.conversations, conversationID)
}

// ============ 使用 Memory ============
type AgentWithMemory struct {
    agent  *react.Agent
    memory *SimpleMemory
}

func NewAgentWithMemory(agent *react.Agent, maxWindow int) *AgentWithMemory {
    return &AgentWithMemory{
        agent:  agent,
        memory: NewSimpleMemory(maxWindow),
    }
}

func (a *AgentWithMemory) Chat(ctx context.Context, conversationID, query string) error {
    // 1. 获取历史
    history := a.memory.GetMessages(conversationID)

    // 2. 添加用户消息
    userMsg := schema.UserMessage(query)
    a.memory.Append(conversationID, userMsg)

    // 3. 构建消息列表
    messages := append(history, userMsg)

    // 4. 执行 Agent
    stream, err := a.agent.Stream(ctx, messages)
    if err != nil {
        return err
    }
    defer stream.Close()

    // 5. 收集响应
    var response string
    for {
        chunk, err := stream.Recv()
        if err != nil {
            break
        }
        response += chunk.Content
        fmt.Print(chunk.Content)
    }

    // 6. 保存 AI 回复
    fmt.Println()
    a.memory.Append(conversationID, schema.AssistantMessage(response, nil))

    return nil
}

func main() {
    ctx := context.Background()

    // 初始化 Agent（简化省略具体代码）
    // agent, _ := react.NewAgent(...)

    agentWithMemory := NewAgentWithMemory(nil, 10)  // 最多保留10条

    // ============ 第一轮对话 ============
    fmt.Println("=== 第一轮 ===")
    agentWithMemory.Chat(ctx, "user1", "我叫张三")

    // ============ 第二轮对话 ============
    fmt.Println("\n=== 第二轮 ===")
    agentWithMemory.Chat(ctx, "user1", "我叫什么名字？")

    // ============ 查看历史 ============
    fmt.Println("\n=== 对话历史 ===")
    history := agentWithMemory.memory.GetMessages("user1")
    for i, msg := range history {
        fmt.Printf("%d. [%s]: %s\n", i+1, msg.Role, msg.Content)
    }
}
```

## Step 8: Chain

### 学习目标
使用 Chain 串联多个组件。

### 什么是 Chain

Chain 将多个组件线性串联，形成处理流水线。

### 完整代码

```go
package main

import (
    "context"
    "fmt"
    "log"

    "github.com/cloudwego/eino/components/model/openai"
    "github.com/cloudwego/eino/components/prompt"
    "github.com/cloudwego/eino/compose"
    "github.com/cloudwego/eino/schema"
)

func main() {
    ctx := context.Background()

    // ============ 1. 创建组件 ============
    model, _ := openai.NewChatModel(ctx, &openai.ChatModelConfig{
        Model:  "gpt-4o-mini",
        APIKey: "your-key",
    })

    // ============ 2. 创建 Prompt 模板 ============
    chatPrompt, err := chatprompt.New(
        // 系统消息（可以引用变量）
        &schema.Message{
            Role:       schema.System,
            Content:    "你是一个{role}助手，请用{style}风格回答用户问题。",
        },
        // 消息占位符
        chatprompt.MessagesByUser(),
    )
    if err != nil {
        log.Fatal(err)
    }

    // ============ 3. 创建 Chain ============
    chain := compose.NewChain[*schema.Message, *schema.Message]()

    chain.
        // 第一步：处理 Prompt 模板
        AppendChatTemplate(chatPrompt,
            compose.WithNodeName("prompt"),
            compose.WithInputKey("userMessage"),      // 输入 key
            compose.WithOutputKey("messages"),       // 输出 key
            // 填充变量
            compose.WithTemplateVariable("role", "技术"),
            compose.WithTemplateVariable("style", "专业"),
        ).
        // 第二步：调用模型
        AppendChatModel(model,
            compose.WithNodeName("llm"),
        )

    // ============ 4. 编译 Chain ============
    runner, err := chain.Compile(ctx)
    if err != nil {
        log.Fatal("编译失败: ", err)
    }

    // ============ 5. 执行 ============
    input := schema.UserMessage("什么是 Go 语言？")

    output, err := runner.Invoke(ctx, input)
    if err != nil {
        log.Fatal("执行失败: ", err)
    }

    fmt.Printf("回复: %s\n", output.Content)
}
```

## Step 9: Graph

### 学习目标
使用 Graph 实现复杂的 DAG 编排。

### 什么是 Graph

Graph 是有向无环图，支持分支和合并。

### 完整代码

```go
package main

import (
    "context"
    "fmt"
    "log"

    "github.com/cloudwego/eino/components/model/openai"
    "github.com/cloudwego/eino/compose"
    "github.com/cloudwego/eino/schema"
)

func main() {
    ctx := context.Background()

    // ============ 1. 创建组件 ============
    model, _ := openai.NewChatModel(ctx, &openai.ChatModelConfig{
        Model:  "gpt-4o-mini",
        APIKey: "your-key",
    })

    // ============ 2. 创建 Graph ============
    // Graph<I, O> - I 是输入类型，O 是输出类型
    g := compose.NewGraph[string, string]()

    // ============ 3. 添加节点 ============

    // 节点1: 分析用户意图
    g.AddLambdaNode("classifier",
        compose.InvokableLambda(func(ctx context.Context, input string) (string, error) {
            // 简单分类：问技术问题 vs 其他
            techKeywords := []string{"怎么", "如何", "是什么", "为什么"}
            for _, kw := range techKeywords {
                if contains(input, kw) {
                    return "tech", nil
                }
            }
            return "other", nil
        }),
        compose.WithNodeName("classifier"),
    )

    // 节点2: 技术助手
    g.AddLambdaNode("tech_response",
        compose.InvokableLambda(func(ctx context.Context, input string) (string, error) {
            return fmt.Sprintf("[技术助手] 关于「%s」，我来详细解答...", input), nil
        }),
        compose.WithNodeName("tech_response"),
    )

    // 节点3: 通用助手
    g.AddLambdaNode("other_response",
        compose.InvokableLambda(func(ctx context.Context, input string) (string, error) {
            return fmt.Sprintf("[通用助手] 「%s」这个问题，我来帮你分析...", input), nil
        }),
        compose.WithNodeName("other_response"),
    )

    // ============ 4. 添加边 ============
    g.AddEdge(compose.START, "classifier")

    // ============ 5. 添加条件分支 ============
    // 根据 classifier 的输出决定走哪个分支
    branch := compose.NewGraphBranch(
        func(ctx context.Context, classification string) (string, error) {
            return classification, nil  // 直接用输出作为目标节点名
        },
        map[string]bool{
            "tech_response":  true,   // tech 分支结束
            "other_response": true,   // other 分支结束
        },
    )
    g.AddBranch("classifier", branch)

    // 连接分支到结束节点
    g.AddEdge("tech_response", compose.END)
    g.AddEdge("other_response", compose.END)

    // ============ 6. 编译执行 ============
    runner, err := g.Compile(ctx)
    if err != nil {
        log.Fatal(err)
    }

    // 测试技术问题
    fmt.Println("测试技术问题:")
    result1, _ := runner.Invoke(ctx, "Go 语言怎么实现并发？")
    fmt.Printf("结果: %s\n\n", result1)

    // 测试其他问题
    fmt.Println("测试其他问题:")
    result2, _ := runner.Invoke(ctx, "今天天气不错")
    fmt.Printf("结果: %s\n", result2)
}

func contains(s, substr string) bool {
    return len(s) >= len(substr) && (s == substr || len(s) > 0 && containsHelper(s, substr))
}

func containsHelper(s, substr string) bool {
    for i := 0; i <= len(s)-len(substr); i++ {
        if s[i:i+len(substr)] == substr {
            return true
        }
    }
    return false
}
```

### Graph 执行流程

```
┌─────────────────────────────────────────────────────────────────┐
│                      Graph 执行流程                               │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  START ──► [classifier] ──┬──► "tech" ──► [tech_response] ──► END
│                          │
│                          └──► "other" ──► [other_response] ──► END
│                                                                  │
```

### Graph vs Chain

| 特性 | Chain | Graph |
|------|-------|-------|
| 结构 | 线性 | 有向无环图 |
| 分支 | 不支持 | 支持 |
| 合并 | 不支持 | 支持 |
| 条件路由 | 不支持 | 支持 |

2. 创建一个并行 Graph：同时执行多个搜索，取最快的结果

## Step 10: HTTP 服务

### 学习目标
将 Agent 封装成 HTTP API 服务。

### 完整代码

```go
package main

import (
    "context"
    "errors"
    "fmt"
    "log"
    "net/http"
    "time"

    "github.com/cloudwego/eino/flow/agent/react"
    "github.com/cloudwego/eino/schema"
    "github.com/cloudwego/kitex-contrib/callback-trigger/httptrigger"

    "github.com/cloudwego/hertz/pkg/app"
    "github.com/cloudwego/hertz/pkg/app/server"
   "github.com/cloudwego/hertz/pkg/common/json"
    "github.com/cloudwego/hertz/pkg/protocol/consts"
)

// Memory 存储（简化版）
type InMemoryStore struct {
    conversations map[string][]*schema.Message
}

func NewInMemoryStore() *InMemoryStore {
    return &InMemoryStore{
        conversations: make(map[string][]*schema.Message),
    }
}

func (s *InMemoryStore) Get(id string) []*schema.Message {
    return s.conversations[id]
}

func (s *InMemoryStore) Append(id string, msgs ...*schema.Message) {
    s.conversations[id] = append(s.conversations[id], msgs...)
}

// ============ HTTP Handler ============
type AgentHandler struct {
    agent *react.Agent
    store *InMemoryStore
}

func NewAgentHandler(agent *react.Agent) *AgentHandler {
    return &AgentHandler{
        agent: agent,
        store: NewInMemoryStore(),
    }
}

type ChatRequest struct {
    ID   string `json:"id"`   // 会话 ID
    Msg  string `json:"msg"`  // 消息内容
    NewChat bool `json:"new_chat"`  // 是否新建会话
}

func (h *AgentHandler) Chat(c context.Context, ctx *app.RequestContext) {
    var req ChatRequest
    if err := json.Unmarshal(ctx.Body(), &req); err != nil {
        ctx.JSON(consts.StatusBadRequest, map[string]string{"error": "invalid request"})
        return
    }

    // 新建会话
    if req.NewChat {
        delete(h.store.conversations, req.ID)
    }

    // 获取历史
    history := h.store.Get(req.ID)

    // 添加用户消息
    userMsg := schema.UserMessage(req.Msg)
    h.store.Append(req.ID, userMsg)

    // 执行 Agent
    stream, err := h.agent.Stream(c, append(history, userMsg))
    if err != nil {
        ctx.JSON(consts.StatusInternalServerError, map[string]string{"error": err.Error()})
        return
    }
    defer stream.Close()

    // 设置 SSE 头
    ctx.SetStatusCode(consts.StatusOK)
    ctx.Response.Header.Set("Content-Type", "text/event-stream")
    ctx.Response.Header.Set("Cache-Control", "no-cache")

    // 流式返回
    var fullResponse string
    for {
        chunk, err := stream.Recv()
        if errors.Is(err, io.EOF) {
            break
        }
        if err != nil {
            fmt.Printf("stream error: %v\n", err)
            break
        }

        fullResponse += chunk.Content
        // 发送 SSE 事件
        ctx.JSON(200, map[string]string{"content": chunk.Content})
        ctx.Flush()
    }

    // 保存 AI 回复
    h.store.Append(req.ID, schema.AssistantMessage(fullResponse, nil))

    // 发送完成信号
    ctx.JSON(200, map[string]string{"done": "true"})
    ctx.Flush()
}

func main() {
    ctx := context.Background()

    // ============ 1. 初始化 Agent ============
    // 简化省略，参考 Step 6
    agent, _ := react.NewAgent(ctx, &react.AgentConfig{...})

    // ============ 2. 创建 HTTP Server ============
    h := server.Default()

    // 注册路由
    handler := NewAgentHandler(agent)

    h.POST("/chat", handler.Chat)
    h.GET("/health", func(ctx context.Context, c *app.RequestContext) {
        c.JSON(200, map[string]string{"status": "ok"})
    })

    // ============ 3. 启动服务 ============
    fmt.Println("服务启动: http://localhost:8888")
    fmt.Println("测试: curl -X POST http://localhost:8888/chat \\")
    fmt.Println("  -H 'Content-Type: application/json' \\")
    fmt.Println("  -d '{\"id\":\"test\",\"msg\":\"你好\"}'")

    if err := h.Run(); err != nil {
        log.Fatal(err)
    }
}
```
