## 1. 基础模板

### 1.1 最小 ChatModel 调用

```go
ctx := context.Background()

// 创建模型
model, _ := ark.NewChatModel(ctx, &ark.ChatModelConfig{
    Model:  os.Getenv("ARK_MODEL"),
    APIKey: os.Getenv("ARK_API_KEY"),
})

// 调用
messages := []*schema.Message{
    schema.SystemMessage("你是一个有帮助的助手"),
    schema.UserMessage("你好"),
}

msg, err := model.Generate(ctx, messages)
fmt.Println(msg.Content)
```

### 1.2 流式调用

```go
stream, err := model.Stream(ctx, messages)
defer stream.Close()

for {
    chunk, err := stream.Recv()
    if errors.Is(err, io.EOF) {
        break
    }
    if err != nil {
        log.Fatal(err)
    }
    fmt.Print(chunk.Content)
}
```

```go
tool, err := utils.InferTool(

        return doSearch(req.Query)
    },

```

### 3.2 带参数验证的工具

```go
tool, err := utils.InferTool(
    "calculator",
    "执行数学计算",
    func(ctx context.Context, req *CalcRequest) (string, error) {
        if req.A < 0 || req.B < 0 {
            return "", errors.New("参数必须为正数")
        }
        return fmt.Sprintf("%d", req.A + req.B), nil
    },
)

type CalcRequest struct {
    A int `json:"a" jsonschema_description:"第一个数"`
    B int `json:"b" jsonschema_description:"第二个数"`
}
```

## 6. Graph

### 6.1 简单 Graph

```go
g := compose.NewGraph[*Input, *Output]()

g.AddLambdaNode("processor", processorLambda)
g.AddChatModelNode("llm", model)
g.AddLambdaNode("formatter", formatterLambda)

g.AddEdge(compose.START, "processor")
g.AddEdge("processor", "llm")
g.AddEdge("llm", "formatter")
g.AddEdge("formatter", compose.END)

runner, err := g.Compile(ctx)
```

### 6.2 带分支的 Graph

```go
// 条件分支函数
check := func(ctx context.Context, msg *Message) (string, error) {
    if len(msg.ToolCalls) > 0 {
        return "tools", nil
    }
    return "end", nil
}

branch := compose.NewStreamGraphBranch(check,
    map[string]bool{
        "tools": true,
        "end":   true,
    })

g.AddBranch("llm", branch)
g.AddEdge("tools", "llm")  // 工具执行后回到 LLM
```

## 8. 流式处理

### 8.1 SSE HTTP 服务

```go
h.GET("/chat", func(ctx context.Context, c *app.RequestContext) {
    stream, _ := agent.Stream(ctx, messages)

    c.SetStatusCode(200)
    c.Response.Header.Set("Content-Type", "text/event-stream")

    s := sse.NewStream(c)
    for {
        chunk, err := stream.Recv()
        if errors.Is(err, io.EOF) {
            break
        }
        s.Publish(&sse.Event{Data: []byte(chunk.Content)})
    }
})
```

### 8.2 流复制

```go
stream, _ := agent.Stream(ctx, messages)
srs := stream.Copy(2)

// goroutine 保存记忆
go func() {
    fullMsgs := make([]*schema.Message, 0)
    for {
        chunk, err := srs[1].Recv()
        if errors.Is(err, io.EOF) {
            break
        }
        fullMsgs = append(fullMsgs, chunk)
    }
    conversation.Append(fullMsgs)
}()

// 返回流给客户端
return srs[0], nil
```

### 8.3 创建自定义流

```go
sr, sw := schema.Pipe[string](10)  // buffer size

go func() {
    for _, chunk := range chunks {
        sw.Send(chunk)
    }
    sw.Close()
}()

for {
    chunk, err := sr.Recv()
    if errors.Is(err, io.EOF) {
        break
    }
    fmt.Print(chunk)
}
```

## 9. 错误处理

### 9.1 常见错误

```go
// 检查 EOF (流结束)
if errors.Is(err, io.EOF) {
    break
}

// 检查上下文取消
if errors.Is(err, context.Canceled) {
    return
}

// 检查超时
if errors.Is(err, context.DeadlineExceeded) {
    return
}
```

### 9.2 错误恢复

```go
// 中断恢复
events := agent.Run(ctx, input)

for event := range events.Iter() {
    if event.Action != nil && event.Action.Interrupted != nil {
        // 保存恢复点
        checkpoint := event.Action.Interrupted.Checkpoint

        // 用户确认后恢复
        events.Resume(ctx, &ResumeInput{
            Checkpoint: checkpoint,
            Approved:   true,
        })
    }
}
```

```go

var g errgroup.Group
var results []Result

for _, task := range tasks {
    task := task
    g.Go(func() error {
        result, err := process(task)
        if err == nil {
            mu.Lock()
            results = append(results, result)
            mu.Unlock()
        }
        return err
    })
}

}
```
