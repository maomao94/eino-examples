## 1. 概述

### 为什么需要 Memory

| 没有 Memory | 有 Memory |
|-------------|-----------|
| 每次都是新对话 | 记得之前的对话 |
| 无法理解上下文 | 理解对话连续性 |
| 重复说明背景 | 自动记住上下文 |

```
┌─────────────────────────────────────────────────────────────────┐
│                      Memory 工作原理                              │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  用户: "我叫张三"                                                │
│       │                                                         │
│       ▼                                                         │
│  ┌─────────────────────────────────────────────────────────┐    │
│  │  Memory                                                 │    │
│  │  ┌─────────────────────────────────────────────────┐   │    │
│  │  │ [UserMessage("我叫张三")]                        │   │    │
│  │  │ [AssistantMessage("你好张三！")]                 │   │    │
│  │  └─────────────────────────────────────────────────┘   │    │
│  └─────────────────────────────────────────────────────────┘    │
│       │                                                         │

│  用户: "我叫什么名字？"                                          │

│                                                                  │
└─────────────────────────────────────────────────────────────────┘
```

### Memory 接口

Eino 没有强制的 Memory 接口，通常的做法是：

```go
// 自定义 Memory 接口
type Memory interface {
    GetMessages(conversationID string) []*schema.Message
    Append(conversationID string, msg *schema.Message)
    Clear(conversationID string)
}
```

## 2. 内存 Memory

### 2.1 简单实现

```go
package main

import (
    "sync"

    "github.com/cloudwego/eino/schema"
)

// ============ 简单内存存储 ============
type InMemoryMemory struct {
    mu           sync.RWMutex
    conversations map[string][]*schema.Message
}

func NewInMemoryMemory() *InMemoryMemory {
    return &InMemoryMemory{
        conversations: make(map[string][]*schema.Message),
    }
}

func (m *InMemoryMemory) GetMessages(conversationID string) []*schema.Message {
    m.mu.RLock()
    defer m.mu.RUnlock()
    return m.conversations[conversationID]
}

func (m *InMemoryMemory) Append(conversationID string, msg *schema.Message) {
    m.mu.Lock()
    defer m.mu.Unlock()
    m.conversations[conversationID] = append(
        m.conversations[conversationID],
        msg,
    )
}

func (m *InMemoryMemory) Clear(conversationID string) {
    m.mu.Lock()
    defer m.mu.Unlock()
    delete(m.conversations, conversationID)
}
```

### 2.2 带滑动窗口

```go
type SlidingWindowMemory struct {
    *InMemoryMemory
    maxWindow int  // 最大保留消息数
}

func NewSlidingWindowMemory(maxWindow int) *SlidingWindowMemory {
    return &SlidingWindowMemory{
        InMemoryMemory: NewInMemoryMemory(),
        maxWindow:     maxWindow,
    }
}

func (m *SlidingWindowMemory) GetMessages(conversationID string) []*schema.Message {
    messages := m.InMemoryMemory.GetMessages(conversationID)

    // 只返回最近的消息
    if len(messages) > m.maxWindow {
        return messages[len(messages)-m.maxWindow:]
    }
    return messages
}
```

### 2.3 使用示例

```go
func main() {
    memory := NewSlidingWindowMemory(10)  // 最多保留10条

    // 添加消息
    memory.Append("user1", schema.UserMessage("你好"))
    memory.Append("user1", schema.AssistantMessage("你好！有什么可以帮助你的？"))
    memory.Append("user1", schema.UserMessage("我叫张三"))

    // 获取历史
    history := memory.GetMessages("user1")
    fmt.Printf("历史消息数: %d\n", len(history))

    // 打印
    for i, msg := range history {
        fmt.Printf("%d. [%s]: %s\n", i+1, msg.Role, msg.Content)
    }
}
```

## 3. 文件持久化

### 3.1 JSONL 存储

```go
package main

import (
    "bufio"
    "context"
    "encoding/json"
    "os"
    "path/filepath"
    "sync"

    "github.com/cloudwego/eino/schema"
)

// ============ 文件 Memory ============
type FileMemory struct {
    mu      sync.RWMutex
    baseDir string
    cache   map[string][]*schema.Message  // 内存缓存
}

func NewFileMemory(baseDir string) (*FileMemory, error) {
    // 创建目录
    if err := os.MkdirAll(baseDir, 0755); err != nil {
        return nil, err
    }

    return &FileMemory{
        baseDir: baseDir,
        cache:   make(map[string][]*schema.Message),
    }, nil
}

// 文件路径
func (m *FileMemory) filePath(conversationID string) string {
    return filepath.Join(m.baseDir, conversationID+".jsonl")
}

// 加载对话
func (m *FileMemory) loadConversation(conversationID string) []*schema.Message {
    filePath := m.filePath(conversationID)

    file, err := os.Open(filePath)
    if err != nil {
        if os.IsNotExist(err) {
            return nil
        }
        return nil
    }
    defer file.Close()

    var messages []*schema.Message
    scanner := bufio.NewScanner(file)

    for scanner.Scan() {
        line := scanner.Bytes()
        var msg schema.Message
        if err := json.Unmarshal(line, &msg); err != nil {
            continue
        }
        messages = append(messages, &msg)
    }

    return messages
}

// 保存对话
func (m *FileMemory) saveConversation(conversationID string, messages []*schema.Message) error {
    filePath := m.filePath(conversationID)

    file, err := os.OpenFile(filePath, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0644)
    if err != nil {
        return err
    }
    defer file.Close()

    encoder := json.NewEncoder(file)
    for _, msg := range messages {
        if err := encoder.Encode(msg); err != nil {
            return err
        }
    }

    return nil
}

// GetMessages
func (m *FileMemory) GetMessages(conversationID string) []*schema.Message {
    m.mu.RLock()
    defer m.mu.RUnlock()

    // 先从缓存获取
    if msgs, ok := m.cache[conversationID]; ok {
        return msgs
    }

    // 从文件加载
    msgs := m.loadConversation(conversationID)
    m.cache[conversationID] = msgs
    return msgs
}

// Append
func (m *FileMemory) Append(conversationID string, msg *schema.Message) {
    m.mu.Lock()
    defer m.mu.Unlock()

    // 添加到缓存
    m.cache[conversationID] = append(m.cache[conversationID], msg)

    // 追加到文件
    filePath := m.filePath(conversationID)
    file, err := os.OpenFile(filePath, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0644)
    if err != nil {
        return
    }
    defer file.Close()

    encoder := json.NewEncoder(file)
    encoder.Encode(msg)
}

// Clear
func (m *FileMemory) Clear(conversationID string) {
    m.mu.Lock()
    defer m.mu.Unlock()

    delete(m.cache, conversationID)
    os.Remove(m.filePath(conversationID))
}
```

### 3.2 使用示例

```go
func main() {
    // 创建文件 Memory
    memory, err := NewFileMemory("./data/conversations")
    if err != nil {
        panic(err)
    }

    // 添加消息
    memory.Append("user1", schema.UserMessage("你好"))
    memory.Append("user1", schema.AssistantMessage("你好！"))
    memory.Append("user1", schema.UserMessage("我叫张三"))

    // 重启后仍能获取
    history := memory.GetMessages("user1")
    fmt.Println("历史消息:")
    for _, msg := range history {
        fmt.Printf("  [%s]: %s\n", msg.Role, msg.Content)
    }
}
```

## 4. Redis Memory

### 4.1 Redis 实现

```go
package main

import (
    "context"
    "encoding/json"
    "fmt"
    "time"

    "github.com/redis/go-redis/v9"
    "github.com/cloudwego/eino/schema"
)

// ============ Redis Memory ============
type RedisMemory struct {
    client *redis.Client
    prefix string
    ttl    time.Duration  // 过期时间
}

func NewRedisMemory(client *redis.Client, prefix string, ttl time.Duration) *RedisMemory {
    return &RedisMemory{
        client: client,
        prefix: prefix,
        ttl:    ttl,
    }
}

func (m *RedisMemory) key(conversationID string) string {
    return m.prefix + ":" + conversationID
}

func (m *RedisMemory) GetMessages(conversationID string) []*schema.Message {
    ctx := context.Background()

    data, err := m.client.Get(ctx, m.key(conversationID)).Bytes()
    if err != nil {
        if err == redis.Nil {
            return nil
        }
        return nil
    }

    var messages []*schema.Message
    if err := json.Unmarshal(data, &messages); err != nil {
        return nil
    }

    return messages
}

func (m *RedisMemory) Append(conversationID string, msg *schema.Message) {
    ctx := context.Background()

    // 获取现有消息
    messages := m.GetMessages(conversationID)
    messages = append(messages, msg)

    // 保存
    data, _ := json.Marshal(messages)
    m.client.Set(ctx, m.key(conversationID), data, m.ttl)
}

func (m *RedisMemory) Clear(conversationID string) {
    ctx := context.Background()
    m.client.Del(ctx, m.key(conversationID))
}
```

### 4.2 使用示例

```go
func main() {
    // 创建 Redis 客户端
    client := redis.NewClient(&redis.Options{
        Addr: "localhost:6379",
    })

    // 创建 Redis Memory
    memory := NewRedisMemory(client, "chat", 24*time.Hour)

    // 使用
    memory.Append("user1", schema.UserMessage("你好"))
    memory.Append("user1", schema.AssistantMessage("你好！"))

    history := memory.GetMessages("user1")
    fmt.Println("历史消息:", history)
}
```

## 5. 滑动窗口

### 5.1 为什么需要滑动窗口

```
问题场景:
用户进行100轮对话，每轮10条消息
总共1000条消息 → Token超限！

解决方案: 滑动窗口
只保留最近的消息，比如最近20条
早期消息被自动丢弃
```

### 5.2 滑动窗口实现

```go
type SlidingWindowMemory struct {
    Memory
    maxWindow int
}

func NewSlidingWindowMemory(memory Memory, maxWindow int) *SlidingWindowMemory {
    return &SlidingWindowMemory{
        Memory:    memory,
        maxWindow: maxWindow,
    }
}

func (m *SlidingWindowMemory) GetMessages(conversationID string) []*schema.Message {
    allMessages := m.Memory.GetMessages(conversationID)

    // 只返回最近的
    if len(allMessages) > m.maxWindow {
        return allMessages[len(allMessages)-m.maxWindow:]
    }
    return allMessages
}
```

### 5.3 Token 感知滑动窗口

```go
type TokenAwareMemory struct {
    *SlidingWindowMemory
    maxTokens int
    tokenizer func(text string) int  // Tokenizer 函数
}

func (m *TokenAwareMemory) GetMessages(conversationID string) []*schema.Message {
    allMessages := m.Memory.GetMessages(conversationID)

    // 从最新消息开始，往前计算 Token
    var windowed []*schema.Message
    tokenCount := 0

    for i := len(allMessages) - 1; i >= 0; i-- {
        msg := allMessages[i]
        tokens := m.tokenizer(msg.Content)

        if tokenCount+tokens > m.maxTokens {
            break
        }

        windowed = append([]*schema.Message{msg}, windowed...)
        tokenCount += tokens
    }

    return windowed
}
```

## 6. 摘要压缩

### 6.1 摘要 Memory

```go
type SummarizingMemory struct {
    Memory
    model    model.ChatModel
    threshold int  // 超过多少条消息后摘要
    summary   map[string]*schema.Message  // 摘要缓存
}

func NewSummarizingMemory(memory Memory, model model.ChatModel, threshold int) *SummarizingMemory {
    return &SummarizingMemory{
        Memory:    memory,
        model:     model,
        threshold: threshold,
        summary:   make(map[string]*schema.Message),
    }
}

func (m *SummarizingMemory) GetMessages(conversationID string) []*schema.Message {
    messages := m.Memory.GetMessages(conversationID)

    // 如果超过阈值，生成摘要
    if len(messages) > m.threshold && m.summary[conversationID] == nil {
        summary := m.summarize(conversationID, messages[:len(messages)/2])
        m.summary[conversationID] = summary
        messages = append([]*schema.Message{summary}, messages[m.threshold/2:]...)
    }

    return messages
}

func (m *SummarizingMemory) summarize(conversationID string, messages []*schema.Message) *schema.Message {
    ctx := context.Background()

    prompt := "请用几句话总结以下对话的要点：\n"
    for _, msg := range messages {
        prompt += fmt.Sprintf("%s: %s\n", msg.Role, msg.Content)
    }

    resp, _ := m.model.Generate(ctx, []*schema.Message{
        schema.SystemMessage("你是一个对话摘要助手，请简洁总结。"),
        schema.UserMessage(prompt),
    })

    return schema.AssistantMessage(resp.Content, nil)
}
```

### 6.2 使用示例

```go
func main() {
    baseMemory := NewInMemoryMemory()

    // 创建带摘要的 Memory
    memory := NewSummarizingMemory(baseMemory, model, 20)  // 超过20条消息后摘要

    // 自动处理
    for i := 0; i < 50; i++ {
        memory.Append("user1", schema.UserMessage(fmt.Sprintf("消息 %d", i)))
        memory.Append("user1", schema.AssistantMessage(fmt.Sprintf("回复 %d", i)))
    }

    // 获取消息（会自动摘要早期对话）
    messages := memory.GetMessages("user1")
    fmt.Printf("消息数: %d (原始100条)\n", len(messages))
}
```

## 7. 最佳实践

### 7.1 选择存储方式

| 场景 | 推荐方案 |
|------|---------|
| 单机开发测试 | 内存 Memory |
| 小规模生产 | 文件 Memory (JSONL) |
| 分布式部署 | Redis Memory |
| 超长对话 | 摘要压缩 |

### 7.2 Token 控制

```go
// 建议配置
const (
    MaxMessages = 20        // 最大消息数
    MaxTokens   = 8000      // 最大 Token（留余量给输出）
)
```

### 7.3 清理策略

```go
// 方法1: 定时清理
go func() {
    ticker := time.NewTicker(24 * time.Hour)
    for range ticker.C {
        memory.ClearExpired()
    }
}()

// 方法2: 懒清理
func (m *FileMemory) lazyCleanup() {
    messages := m.GetMessages(conversationID)
    if len(messages) > MaxMessages {
        messages = messages[len(messages)-MaxMessages:]
        m.saveConversation(conversationID, messages)
    }
}
```

### 7.4 完整对话管理

```go
type ConversationManager struct {
    memory   Memory
    maxWindow int
}

func (cm *ConversationManager) Chat(conversationID, query string) (string, error) {
    // 1. 获取历史
    history := cm.memory.GetMessages(conversationID)

    // 2. 添加用户消息
    userMsg := schema.UserMessage(query)
    cm.memory.Append(conversationID, userMsg)

    // 3. 调用 Agent
    resp := agent.Chat(ctx, append(history, userMsg))

    // 4. 保存回复
    cm.memory.Append(conversationID, resp)

    return resp.Content, nil
}

func (cm *ConversationManager) NewConversation(conversationID string) {
    cm.memory.Clear(conversationID)
}
```
