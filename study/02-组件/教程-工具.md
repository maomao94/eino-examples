## 1. 概述

### 什么是 Tool

Tool（工具）是让 AI 能够执行实际操作的能力扩展。

```
┌─────────────────────────────────────────────────────────────────┐
│                      Tool 在 Agent 中的位置                        │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│                        ┌─────────────┐                          │
│                        │   Agent    │                          │
│                        └──────┬──────┘                          │
│                               │                                   │
│      ┌────────────────────────┼────────────────────────┐        │
│      │                        │                        │        │
│      ▼                        ▼                        ▼        │
│  ┌────────┐            ┌────────┐            ┌────────┐      │
│  │  LLM   │            │ Memory │            │ Tools  │      │
│  │ (大脑) │            │ (记忆) │            │ (手脚) │      │
│  └────────┘            └────────┘            └────────┘      │
│                                                     │            │
│                        ┌────────────────────────────┼─────────┐ │
│                        │                            ▼         │ │
│                        │  ┌────────┐ ┌────────┐ ┌────────┐  │ │
│                        │  │ Search │ │ Calc   │ │ File   │  │ │
│                        │  └────────┘ └────────┘ └────────┘  │ │
│                        └─────────────────────────────────────────┘ │
└─────────────────────────────────────────────────────────────────┘
```

### Tool 接口

```go
// 工具元数据
type BaseTool interface {
    Info(ctx context.Context) (*schema.ToolInfo, error)
}

// 同步执行
type InvokableTool interface {
    BaseTool
    InvokableRun(ctx context.Context, argumentsInJSON string,
                 opts ...Option) (string, error)
}

// 流式执行
type StreamableTool interface {
    BaseTool
    StreamableRun(ctx context.Context, argumentsInJSON string,
                  opts ...Option) (*schema.StreamReader[string], error)
}
```

## 2. 快速开始

### 最简单的工具

```go
package main

import (
    "context"
    "fmt"

    "github.com/cloudwego/eino/components/tool/utils"
)

// 定义请求结构
type HelloRequest struct {
    Name string `json:"name" jsonschema_description:"问候的姓名"`
}

func main() {
    // 使用 InferTool 创建工具
    tool, err := utils.InferTool(
        "hello",                    // 工具名称
        "向某人打招呼",               // 工具描述
        func(ctx context.Context, req *HelloRequest) (string, error) {
            return fmt.Sprintf("你好，%s！", req.Name), nil
        },
    )
    if err != nil {
        panic(err)
    }

    // 获取工具信息
    info, _ := tool.Info(context.Background())
    fmt.Printf("工具名称: %s\n", info.Name)
    fmt.Printf("工具描述: %s\n", info.Desc)
}
```

### 带参数的工具

```go
type CalculatorRequest struct {
    A int    `json:"a" jsonschema_description:"第一个数字"`
    B int    `json:"b" jsonschema_description:"第二个数字"`
    Op string `json:"op" jsonschema_description:"操作符：add, sub, mul, div"`
}

calculatorTool, err := utils.InferTool(
    "calculator",
    "执行数学计算",
    func(ctx context.Context, req *CalculatorRequest) (string, error) {
        var result int
        switch req.Op {
        case "add": result = req.A + req.B
        case "sub": result = req.A - req.B
        case "mul": result = req.A * req.B
        case "div":
            if req.B == 0 {
                return "错误：除数不能为零", nil
            }
            result = req.A / req.B
        default:
            return fmt.Sprintf("未知操作: %s", req.Op), nil
        }
        return fmt.Sprintf("%d", result), nil
    },
)
```

## 3. InferTool 详解

### 3.1 基本用法

```go
utils.InferTool(
    name,        // 工具名称
    description, // 描述（会被加入 AI 的提示词）
    implFunc,    // 实现函数
)
```

### 3.2 自动生成 Schema

```go
// InferTool 会自动从请求结构生成 JSON Schema
type Request struct {
    Name  string `json:"name"`
    Age   int    `json:"age"`
    Email string `json:"email" jsonschema_description:"邮箱地址"`
    Phone string `json:"phone,omitempty"`  // omitempty = 可选
}

// 自动生成的 Schema：
// {
//   "type": "object",
//   "properties": {
//     "name": {"type": "string"},
//     "age": {"type": "integer"},
//     "email": {"type": "string", "description": "邮箱地址"},
//     "phone": {"type": "string"}
//   },
//   "required": ["name", "age", "email"]
// }
```

### 3.3 支持的类型

```go
type MyRequest struct {
    // 基础类型
    Name    string  `json:"name"`
    Age     int     `json:"age"`
    Price   float64 `json:"price"`
    Valid   bool    `json:"valid"`

    // 可选字段
    Nickname string `json:"nickname,omitempty"`

    // 带描述
    Email string `json:"email" jsonschema_description:"邮箱地址"`

    // 嵌套对象
    Address struct {
        City    string `json:"city"`
        Street  string `json:"street"`
    } `json:"address"`

    // 数组
    Tags []string `json:"tags"`

    // 枚举
    Status string `json:"status" jsonschema_enum:"active,pending,completed"`
}
```

### 3.4 返回值处理

```go
// 工具返回 string，AI 会看到这个结果
tool, err := utils.InferTool(
    "get_weather",
    "获取天气信息",
    func(ctx context.Context, req *WeatherRequest) (string, error) {
        // 返回字符串结果
        return "北京：晴，28度", nil
    },
)

// 如果返回 error，工具调用失败
func(ctx context.Context, req *Request) (string, error) {
    if req.ID == "" {
        return "", errors.New("ID 不能为空")
    }
    return doSomething(req), nil
}
```

## 4. 完整接口实现

### 4.1 实现 BaseTool + InvokableTool

```go
package main

import (
    "context"
    "encoding/json"
    "fmt"

    "github.com/cloudwego/eino/components/tool"
    "github.com/cloudwego/eino/schema"
)

// ============ 自定义工具 ============
type MyTool struct {
    name string
    desc string
}

func NewMyTool() *MyTool {
    return &MyTool{
        name: "my_tool",
        desc: "我的自定义工具",
    }
}

// Info: 返回工具元数据
func (t *MyTool) Info(ctx context.Context) (*schema.ToolInfo, error) {
    return &schema.ToolInfo{
        Name: t.name,
        Desc: t.desc,
        ArgsSchema: &MyRequest{},
    }, nil
}

// InvokableRun: 执行工具
func (t *MyTool) InvokableRun(ctx context.Context, argsInJSON string, opts ...Option) (string, error) {
    // 解析参数
    var req MyRequest
    if err := json.Unmarshal([]byte(argsInJSON), &req); err != nil {
        return "", fmt.Errorf("参数解析失败: %w", err)
    }

    // 执行业务逻辑
    result, err := t.doSomething(req)
    if err != nil {
        return "", err
    }

    return result, nil
}

// ============ 请求结构 ============
type MyRequest struct {
    Name string `json:"name"`
    Type string `json:"type"`
}

// ============ 使用工具 ============
func main() {
    t := NewMyTool()

    info, _ := t.Info(context.Background())
    fmt.Printf("工具: %s - %s\n", info.Name, info.Desc)

    result, err := t.InvokableRun(context.Background(), `{"name":"test","type":"A"}`)
    if err != nil {
        panic(err)
    }
    fmt.Println("结果:", result)
}
```

### 4.2 流式工具

```go
// 实现 StreamableTool 接口
type StreamingTool struct {
    *MyTool  // 嵌入
}

func NewStreamingTool() *StreamingTool {
    return &StreamingTool{
        MyTool: NewMyTool(),
    }
}

func (t *StreamingTool) StreamableRun(
    ctx context.Context,
    argsInJSON string,
    opts ...Option,
) (*schema.StreamReader[string], error) {

    // 解析参数
    var req MyRequest
    json.Unmarshal([]byte(argsInJSON), &req)

    // 创建管道
    sr, sw := schema.Pipe[string](10)

    // 异步写入
    go func() {
        // 分段发送结果
        chunks := splitResult(req.Name)
        for _, chunk := range chunks {
            sw.Send(chunk)
        }
        sw.Close()
    }()

    return sr, nil
}

// 工具可以同时实现 InvokableRun 和 StreamableRun
// AI 会根据情况选择
```

### 4.3 增强版工具

```go
// 增强版：支持多模态返回
type EnhancedTool struct{}

func (t *EnhancedTool) Info(ctx context.Context) (*schema.ToolInfo, error) {
    return &schema.ToolInfo{
        Name: "enhanced_tool",
        Desc: "增强版工具",
        ArgsSchema: &EnhancedRequest{},
    }, nil
}

func (t *EnhancedTool) InvokableRun(
    ctx context.Context,
    toolArgument *schema.ToolArgument,
    opts ...Option,
) (*schema.ToolResult, error) {

    // 可以返回多模态内容
    return &schema.ToolResult{
        Content: "文本结果",
        Parts: []schema.MessageOutputPart{
            &schema.TextContent{Text: "文本"},
            &schema.ImageURLContent{URL: "https://..."},
        },
    }, nil
}
```

## 5. 工具中间件

### 5.1 错误处理中间件

```go
import "github.com/cloudwego/eino/components/tool/middlewares/errorremover"

errHandler := &errorremover.ErrorRemover{
    RemoveErrorContent: true,  // 移除错误信息
    Replacement:       "操作失败，请重试",
}

tool, _ := utils.InvokableToolToMiddlewares(
    baseTool,
    errHandler,
)
```

### 5.2 JSON 修复中间件

```go
import "github.com/cloudwego/eino/components/tool/middlewares/jsonfix"

jsonFixer := &jsonfix.JSONFixer{
    MaxRetries: 3,
}

tool, _ := utils.InvokableToolToMiddlewares(
    baseTool,
    jsonFixer,
)
```

### 5.3 自定义中间件

```go
// 创建中间件
type LogMiddleware struct{}

func (m *LogMiddleware) BeforeInvoke(ctx context.Context, args []any) (context.Context, []any, error) {
    fmt.Printf("[Tool] 调用前，参数: %v\n", args)
    return ctx, args, nil
}

func (m *LogMiddleware) AfterInvoke(ctx context.Context, result string) (context.Context, string, error) {
    fmt.Printf("[Tool] 调用后，结果长度: %d\n", len(result))
    return ctx, result, nil
}

// 应用中间件
wrapped := compose.InvokableToolMiddleware(logMiddleware)(baseTool)
```

### 5.4 工具节点中间件

```go
// 在 Graph/Agent 中配置
compose.ToolsNodeConfig{
    Tools: []tool.BaseTool{tool1, tool2},
    ToolCallMiddlewares: []compose.ToolMiddleware{
        // 重试中间件
        compose.ToolMiddleware(&RetryMiddleware{MaxRetries: 3}),
        // 限流中间件
        compose.ToolMiddleware(&RateLimitMiddleware{Rate: 10}),
    },
}
```

### 6.3 DuckDuckGo 完整示例

```go
package main

import (
    "context"
    "fmt"
    "log"

    "github.com/cloudwego/eino/components/tool/duckduckgo"
    "github.com/cloudwego/eino/components/tool/utils"
)

func main() {
    ctx := context.Background()

    // 创建搜索工具
    searchTool, err := duckduckgo.NewTextSearchTool(ctx, &duckduckgo.Config{
        Timeout: 30 * time.Second,
    })
    if err != nil {
        log.Fatal(err)
    }

    // 包装为 InferTool 格式
    webSearchTool, err := utils.InferTool(
        "web_search",
        "搜索互联网获取最新信息",
        func(ctx context.Context, req *SearchRequest) (string, error) {
            return searchTool.InvokableRun(ctx, req.Query)
        },
    )

    // 测试
    result, err := webSearchTool.InvokableRun(ctx, `{"query":"Go语言最新版本"}`)
    if err != nil {
        log.Fatal(err)
    }
    fmt.Println("搜索结果:", result)
}

type SearchRequest struct {
    Query string `json:"query"`
}
```

## 7. 最佳实践

### 7.1 工具命名

```go
// ✅ 好命名
utils.InferTool("get_weather", "...", ...)
utils.InferTool("calculate", "...", ...)
utils.InferTool("search_codebase", "...", ...)

// ❌ 差命名
utils.InferTool("tool1", "...", ...)
utils.InferTool("func", "...", ...)
```

### 7.2 描述编写

```go
// ✅ 清晰描述
utils.InferTool(
    "get_weather",
    "获取指定城市的天气信息，包括温度、湿度、风力等",
    ...
)

// ❌ 模糊描述
utils.InferTool(
    "get_weather",
    "获取天气",
    ...
)
```

### 7.3 参数验证

```go
utils.InferTool(
    "transfer_money",
    "转账",
    func(ctx context.Context, req *TransferRequest) (string, error) {
        // 参数验证
        if req.Amount <= 0 {
            return "", errors.New("金额必须大于0")
        }
        if req.Amount > 100000 {
            return "", errors.New("单笔转账限额10万")
        }
        if req.Account == "" {
            return "", errors.New("账户不能为空")
        }

        // 执行业务
        return doTransfer(req)
    },
)

type TransferRequest struct {
    From     string  `json:"from"`
    To       string  `json:"to"`
    Amount   float64 `json:"amount"`
    Currency string  `json:"currency"`
}
```

### 7.4 错误处理

```go
utils.InferTool(
    "call_api",
    "调用外部API",
    func(ctx context.Context, req *APIRequest) (string, error) {
        resp, err := http.Post(req.URL, req.Body)
        if err != nil {
            // 区分错误类型
            if os.IsTimeout(err) {
                return "", errors.New("请求超时，请重试")
            }
            if os.IsNotExist(err) {
                return "", errors.New("服务不可用")
            }
            return "", err
        }
        defer resp.Body.Close()

        if resp.StatusCode != 200 {
            return "", fmt.Errorf("API返回错误: %d", resp.StatusCode)
        }

        body, _ := io.ReadAll(resp.Body)
        return string(body), nil
    },
)
```

### 7.5 工具组合

```go
// 创建多个工具
tools := []tool.BaseTool{
    searchTool,
    calculatorTool,
    fileTool,
    httpTool,
}

// 在 Agent 中使用
agent, _ := react.NewAgent(ctx, &react.AgentConfig{
    Model: model,
    ToolsConfig: compose.ToolsNodeConfig{
        Tools: tools,
        // 并行或顺序执行
        ExecuteSequentially: false,  // true = 顺序，false = 并行
    },
})
```
