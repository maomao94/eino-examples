# 快速开始

本文档帮助你在 5 分钟内跑通 Eino 的第一个程序。

---

## 环境准备

### 前置要求

- Go 1.21 或更高版本
- 网络访问（下载依赖和调用 LLM API）

### 安装 Eino

```bash
go get github.com/cloudwego/eino@latest
go get github.com/cloudwego/eino-ext/...@latest
```

## 第一步：调用模型

最简单的 Eino 程序：调用大语言模型生成回复。

```go
package main

import (
	"context"
	"fmt"
	"github.com/cloudwego/eino/chatbot"
	"github.com/cloudwego/eino/chatbot/completion"
)

// 本示例展示 Eino 的核心使用模式
// 完整示例参考 Eino 官方示例中的 compose/chat 目录

func main() {
	ctx := context.Background()

	// 1. 创建 ChatModel 实例（以 OpenAI 为例）
	model, err := openai.NewChatModel(ctx, &openai.ChatModelConfig{
		APIKey: "your-api-key",
		Model:  "gpt-4o-mini",
	})
	if err != nil {
		panic(err)
	}

	// 2. 发送消息并获取回复
	msgs := []chatbot.Message{
		chatbot.UserMessage("Hello, who are you?"),
	}

	resp, err := model.Generate(ctx, msgs)
	if err != nil {
		panic(err)
	}

	fmt.Println("AI 回复:", resp.Content())
}
```

## 第二步：创建工具

让 AI 能够调用外部函数执行特定任务。

```go
package main

import (
	"context"
	"fmt"
	"github.com/cloudwego/eino/agent"
	"github.com/cloudwego/eino/agent/plan_et"
)

// 定义一个简单的天气查询工具
type WeatherTool struct{}

func (t *WeatherTool) Info() *agent.ToolInfo {
	return &agent.ToolInfo{
		Name:        "get_weather",
		Description: "查询指定城市的天气信息",
		Parameters: map[string]interface{}{
			"type": "object",
			"properties": map[string]interface{}{
				"city": map[string]interface{}{
					"type":        "string",
					"description": "城市名称，如：北京、上海",
				},
			},
			"required": []string{"city"},
		},
	}
}

func (t *WeatherTool) Invoke(ctx context.Context, params map[string]any) (string, error) {
	city, ok := params["city"].(string)
	if !ok {
		return "", fmt.Errorf("invalid city parameter")
	}
	return fmt.Sprintf("%s 今天天气晴朗，温度 25°C", city), nil
}
```

## 第三步：构建 ReAct Agent

将模型和工具组合成自主工作的 Agent。

```go
package main

import (
	"context"
	"fmt"
	"github.com/cloudwego/eino/agent"
	"github.com/cloudwego/eino/agent/react"
	"github.com/cloudwego/eino/chatbot"
	"github.com/cloudwego/eino/chatbot/completion"
)

// 完整的 ReAct Agent 示例
// 参考 Eino 官方示例中的 compose/react 目录

func main() {
	ctx := context.Background()

	// 1. 创建 ChatModel
	model, _ := openai.NewChatModel(ctx, &openai.ChatModelConfig{
		APIKey: "your-api-key",
		Model:  "gpt-4o-mini",
	})

	// 2. 创建工具
	tools := []agent.Tool{
		&WeatherTool{},
	}

	// 3. 创建 ReAct Agent
	agent, err := react.NewAgent(ctx, &react.AgentConfig{
		Model:  model,
		Tools:  tools,
		Prompt: "你是一个有用的助手，可以使用工具来回答问题。",
	})
	if err != nil {
		panic(err)
	}

	// 4. 运行 Agent
	result, err := agent.Run(ctx, "北京现在天气怎么样？")
	if err != nil {
		panic(err)
	}

	fmt.Println("Agent 回复:", result)
}
```

## 第四步：编排工作流

使用 Graph 将多个组件连接成复杂流程。

```go
package main

import (
	"context"
	"fmt"
	"github.com/cloudwego/eino/compose"
)

// 构建一个简单的 RAG 工作流
// 参考 Eino 官方示例中的 compose/graph/rag 目录

func main() {
	ctx := context.Background()

	// 创建各阶段节点
	retriever := compose.NewGraphNode(
		compose.WithNodeName("retriever"),
		compose.WithRetriever(myRetriever),
	)

	generator := compose.NewGraphNode(
		compose.WithNodeName("generator"),
		compose.WithChatModel(myModel),
	)

	// 构建 DAG 图
	graph := compose.NewGraph[*UserMsg, *AIResponse]()

	// 添加节点和边
	graph.AddNode("retrieve", retriever)
	graph.AddNode("generate", generator)
	graph.AddEdge("retrieve", "generate")

	// 编译并运行
	runnable, _ := graph.Compile(ctx)
	result, _ := runnable.Invoke(ctx, &UserMsg{Query: "什么是 Eino?"})
	fmt.Println(result)
}
```
